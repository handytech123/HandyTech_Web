2) Server entry (Express + Socket.IO + OpenAI + Twilio + Email)

Create server/index.js (or merge these parts into your current server). This is JS; if you’re TS, identical code compiles fine.

import 'dotenv/config';
import express from 'express';
import http from 'http';
import { Server } from 'socket.io';
import cors from 'cors';
import nodemailer from 'nodemailer';
import { OpenAI } from 'openai';
import twilio from 'twilio';
import cookieParser from 'cookie-parser';
import csurf from 'csurf';
import path from 'path';
import { fileURLToPath } from 'url';

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

const app = express();
const server = http.createServer(app);

// CORS
const ALLOWED = (process.env.ALLOWED_ORIGINS || '').split(',').filter(Boolean);
app.use(cors({ origin: ALLOWED.length ? ALLOWED : true, credentials: true }));
app.use(express.json());
app.use(express.urlencoded({ extended: true }));
app.use(cookieParser());

// Static (optional)
app.use('/', express.static(path.join(__dirname, '..', 'public')));
app.use('/admin', express.static(path.join(__dirname, '..', 'admin')));

// Socket.IO (respect CORS)
const io = new Server(server, {
  cors: { origin: ALLOWED.length ? ALLOWED : '*', methods: ['GET','POST'], credentials: true }
});

// CSRF (only for state-changing routes)
const csrfProtection = csurf({
  cookie: {
    httpOnly: true,
    sameSite: process.env.CSRF_SAME_SITE || 'lax',
    secure: process.env.NODE_ENV === 'production'
  }
});
app.get('/api/csrf-token', csrfProtection, (req, res) => res.json({ csrfToken: req.csrfToken() }));

// Clients
const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

// Email (optional)
let mailer = null;
if (process.env.SMTP_HOST && process.env.SMTP_USER && process.env.SMTP_PASS && process.env.ADMIN_EMAIL_FROM) {
  mailer = nodemailer.createTransport({
    host: process.env.SMTP_HOST,
    port: Number(process.env.SMTP_PORT || 587),
    secure: Number(process.env.SMTP_PORT || 587) === 465,
    auth: { user: process.env.SMTP_USER, pass: process.env.SMTP_PASS }
  });
}

// Twilio (SMS) (optional but requested)
let smsClient = null;
if (process.env.TWILIO_ACCOUNT_SID && process.env.TWILIO_AUTH_TOKEN && process.env.TWILIO_FROM && process.env.ADMIN_SMS_TO) {
  smsClient = twilio(process.env.TWILIO_ACCOUNT_SID, process.env.TWILIO_AUTH_TOKEN);
}

// Simple store (replace with DB/Redis in prod)
const conversations = new Map(); // convId -> { status, messages, createdAt }

// Helpers
const newId = () => Math.random().toString(36).slice(2) + Date.now().toString(36);

async function notifyAdmin(subject, text) {
  const tasks = [];
  if (mailer && process.env.ADMIN_EMAIL_TO) {
    tasks.push(mailer.sendMail({
      from: process.env.ADMIN_EMAIL_FROM,
      to: process.env.ADMIN_EMAIL_TO,
      subject,
      text
    }));
  }
  if (smsClient) {
    tasks.push(smsClient.messages.create({
      body: `${subject}\n${text}`.slice(0, 1500),
      from: process.env.TWILIO_FROM,
      to: process.env.ADMIN_SMS_TO
    }));
  }
  await Promise.allSettled(tasks);
}

function systemPrompt() {
  return `You are HandyChat for ${process.env.SITE_NAME || 'our company'}.
- Greet, answer simply, and nudge booking when appropriate (2–4 sentences).
- If user wants to book: collect name, phone, address, brief description, preferred time.
- If user asks for a human, say you'll alert someone and request a handoff.`;
}

// Chat endpoint (OpenAI)
app.post('/api/chatbot', csrfProtection, async (req, res) => {
  try {
    const { message, sessionId } = req.body || {};
    if (!message) return res.status(400).json({ error: 'message required' });
    const convId = sessionId || newId();

    if (!conversations.has(convId)) {
      conversations.set(convId, { status: 'bot', messages: [], createdAt: Date.now() });
    }
    const convo = conversations.get(convId);
    convo.messages.push({ role: 'user', content: message, ts: Date.now() });

    // Handoff detection
    if (/human|agent|representative|person|someone|live/i.test(message)) {
      convo.status = 'pending_handoff';
      await notifyAdmin('Handoff requested', `Conversation ${convId} asked for a human.`);
      return res.json({ conversationId: convId, response: "Got it — I’ve alerted a team member to join this chat.", status: 'pending_handoff' });
    }

    // OpenAI reply
    const response = await new OpenAI({ apiKey: process.env.OPENAI_API_KEY })
      .chat.completions.create({
        model: 'gpt-4o-mini',
        temperature: 0.4,
        messages: [
          { role: 'system', content: systemPrompt() },
          ...convo.messages.slice(-12)
        ]
      });

    const text = response.choices?.[0]?.message?.content?.trim() || "Okay!";
    convo.messages.push({ role: 'assistant', content: text, ts: Date.now() });
    res.json({ conversationId: convId, response: text, status: convo.status });
  } catch (e) {
    console.error(e);
    res.status(500).json({ error: 'chatbot failed' });
  }
});

// Appointment endpoint (alerts you by Email/SMS)
app.post('/api/appointment', csrfProtection, async (req, res) => {
  try {
    const { convId, name, phone, email, address, description, preferred } = req.body || {};
    const id = convId || newId();
    const subject = `New Appointment Request (${process.env.SITE_NAME || 'Website'})`;
    const text = `Conversation: ${id}
Name: ${name}
Phone: ${phone}
Email: ${email}
Address: ${address}
Preferred: ${preferred}
Description: ${description}`;
    await notifyAdmin(subject, text);
    res.json({ ok: true });
  } catch (e) {
    console.error(e);
    res.status(500).json({ ok: false, error: 'appointment failed' });
  }
});

// Socket.IO for live handoff (optional UI)
io.on('connection', (socket) => {
  // Keep simple for now; admin UI can be built or reuse existing
});

const PORT = Number(process.env.PORT || 5000);
server.listen(PORT, '0.0.0.0', () => console.log(`HandyChat listening on ${PORT}`));

3) Widget calls (send CSRF & message)

In your widget (e.g., public/widget.js), fetch a CSRF token once and send it on POSTs.

<script>
(async function() {
  let CSRF = "";
  async function getCsrf() {
    const r = await fetch('/api/csrf-token', { credentials: 'include' });
    const j = await r.json(); CSRF = j.csrfToken;
  }
  await getCsrf();

  async function sendMessage(text) {
    const sessionId = localStorage.getItem('chatbot-session-id') ||
      (() => { const s = `sess_${Date.now()}_${Math.random().toString(36).slice(2)}`; localStorage.setItem('chatbot-session-id', s); return s; })();

    const res = await fetch('/api/chatbot', {
      method: 'POST',
      credentials: 'include',
      headers: { 'Content-Type': 'application/json', 'x-csrf-token': CSRF },
      body: JSON.stringify({ message: text, sessionId })
    });

    if (res.status === 403) { await getCsrf(); return sendMessage(text); }
    const data = await res.json();
    // render data.response, watch data.status === 'pending_handoff'
  }

  async function sendAppointment(form) {
    const res = await fetch('/api/appointment', {
      method: 'POST',
      credentials: 'include',
      headers: { 'Content-Type': 'application/json', 'x-csrf-token': CSRF },
      body: JSON.stringify(form)
    });
    if (res.status === 403) { await getCsrf(); return sendAppointment(form); }
    // handle success
  }

  // bind your UI buttons/inputs to sendMessage/sendAppointment
})();
</script>


If your chat UI is React/TS, same logic applies. Use credentials:'include' and add the x-csrf-token header.

6) What Replit will deliver after this

Bot responds via OpenAI (/api/chatbot).

When user types “human”, the server:

Marks the conversation pending_handoff.

Sends SMS via Twilio (and/or email) to you with the conversation ID.

Appointment submissions hit /api/appointment and alert you by SMS/email with all customer details.

7) Common pitfalls (with fixes)

Twilio 21606/20003 errors → wrong from/to numbers or unverified number in trial; verify in Twilio console; ensure E.164 format +1....

No SMS received → check Replit logs for messages.create result; confirm TWILIO_FROM is an active SMS number.

OpenAI 401/429 → wrong key or quota; recheck OPENAI_API_KEY secret.

CSRF 403 → missing credentials:'include' and header; always fetch /api/csrf-token first.

CORS 403 → set ALLOWED_ORIGINS to include the exact Replit preview URL and your site origin.
