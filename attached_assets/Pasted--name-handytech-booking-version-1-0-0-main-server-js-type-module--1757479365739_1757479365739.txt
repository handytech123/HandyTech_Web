import express from "express";
import axios from "axios";
import crypto from "crypto";
import bodyParser from "body-parser";
import path from "path";
import { fileURLToPath } from "url";

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

const app = express();

// Raw body just for webhook signature verification
app.use("/api/calendly/webhook", bodyParser.raw({ type: "*/*" }));
// JSON body for all other routes
app.use(bodyParser.json());

// --- ENV (add in Replit Secrets) ---
const CALENDLY_PAT = process.env.CALENDLY_PAT; // Personal Access Token
let CALENDLY_USER_URI = process.env.CALENDLY_USER_URI || ""; // like https://api.calendly.com/users/XXXX
let CALENDLY_ORG_URI = process.env.CALENDLY_ORG_URI || "";   // like https://api.calendly.com/organizations/XXXX
const CALENDLY_SIGNING_KEY = process.env.CALENDLY_SIGNING_KEY || ""; // from webhook subscription (optional but recommended)

// Static files (your site)
app.use(express.static(path.join(__dirname, "public")));

// Health check
app.get("/health", (_req, res) => res.send("ok"));

// Helper: auth header
const calendly = axios.create({
  baseURL: "https://api.calendly.com",
  headers: { Authorization: `Bearer ${CALENDLY_PAT}` }
});

// Bootstrap to fetch user/org if not provided
async function ensureUserAndOrg() {
  if (!CALENDLY_USER_URI || !CALENDLY_ORG_URI) {
    const { data } = await calendly.get("/users/me");
    CALENDLY_USER_URI = data?.resource?.uri || CALENDLY_USER_URI;
    CALENDLY_ORG_URI = data?.resource?.current_organization || CALENDLY_ORG_URI;
  }
}

// API: list your event types → front end can render buttons dynamically
app.get("/api/event-types", async (_req, res) => {
  try {
    if (!CALENDLY_PAT) {
      return res.status(400).json({ error: "Missing CALENDLY_PAT" });
    }
    await ensureUserAndOrg();

    const { data } = await calendly.get("/event_types", {
      params: { user: CALENDLY_USER_URI }
    });

    const items = (data?.resource || data?.collection || []).map(et => ({
      name: et.name,
      uri: et.uri,
      slug: et.slug,
      scheduling_url: et.scheduling_url,
      duration: et.duration
    }));

    res.json({ user: CALENDLY_USER_URI, organization: CALENDLY_ORG_URI, event_types: items });
  } catch (err) {
    console.error(err?.response?.data || err.message);
    res.status(500).json({ error: "Failed to fetch event types" });
  }
});

// Verify Calendly webhook signature (optional but safer)
function verifyCalendlySignature(signingKey, signatureHeader, bodyBuffer) {
  // Header looks like: "t=TIMESTAMP,v1=HMAC"
  if (!signingKey || !signatureHeader) return false;
  const parts = signatureHeader.split(",").reduce((acc, kv) => {
    const [k, v] = kv.split("=");
    acc[k.trim()] = (v || "").trim();
    return acc;
  }, {});

  const ts = parts["t"];
  const v1 = parts["v1"];
  if (!ts || !v1) return false;

  const payload = `${ts}.${bodyBuffer.toString("utf8")}`;
  const expected = crypto.createHmac("sha256", signingKey).update(payload).digest("hex");

  // Compare safely
  return crypto.timingSafeEqual(Buffer.from(v1), Buffer.from(expected));
}

// Webhook endpoint: receive invitee.created / invitee.canceled
app.post("/api/calendly/webhook", async (req, res) => {
  try {
    const sig = req.get("Calendly-Webhook-Signature");

    if (CALENDLY_SIGNING_KEY) {
      const ok = verifyCalendlySignature(CALENDLY_SIGNING_KEY, sig, req.body);
      if (!ok) {
        console.warn("Invalid Calendly signature");
        return res.status(400).send("Bad signature");
      }
    }

    // Now parse the real JSON after signature check
    const payload = JSON.parse(req.body.toString("utf8"));
    console.log("Calendly webhook event:", payload?.event, payload?.payload?.invitee?.email);

    // TODO: do your automations here:
    // - create job ticket
    // - send SMS
    // - write to a DB
    // For demo, just 200 OK
    res.status(200).send("ok");
  } catch (e) {
    console.error("Webhook error:", e.message);
    res.status(200).send("ok"); // Always 200 to avoid retries during testing
  }
});

// Start server
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log(`HandyTech server running on :${PORT}`));

4) Replit → Secrets (Environment Variables)
Add these in Replit → Tools → Secrets:

CALENDLY_PAT → your personal access token (Calendly → Integrations & Apps → API & Webhooks → Personal access tokens → Generate).

(Optional) CALENDLY_USER_URI and CALENDLY_ORG_URI → if you already know them; otherwise the server fetches them.

(Optional but recommended) CALENDLY_SIGNING_KEY → create this when you make the webhook (below); paste it here so signatures are verified.

5) Create the Webhook Subscription (real-time booking events)
a) Get your org/user URIs (one-time):

bash
Copy code
curl -s https://api.calendly.com/users/me \
  -H "Authorization: Bearer $CALENDLY_PAT" | jq .
Copy:

resource.uri → set as CALENDLY_USER_URI (or let server fetch it)

resource.current_organization → CALENDLY_ORG_URI

b) Create webhook (replace the URL with your deployed endpoint):

bash
Copy code
curl -X POST https://api.calendly.com/webhook_subscriptions \
  -H "Authorization: Bearer $CALENDLY_PAT" \
  -H "Content-Type: application/json" \
  -d '{
    "url": "https://<your-replit-deploy-domain>/api/calendly/webhook",
    "events": ["invitee.created", "invitee.canceled"],
    "organization": "'$CALENDLY_ORG_URI'",
    "scope": "organization",
    "signing_key": true
  }'
Calendly returns a signing_key. Put that value into your Replit Secret CALENDLY_SIGNING_KEY.

6) Run it
Click Run in Replit.

Open /book.html (or make it your homepage).

Book a test slot → you should see a console log from the webhook on your server.
