const fs = require("fs");
const path = require("path");
function loadServicesJSON() {
  const p = path.join(process.cwd(), "server", "data", "services.json");
  return JSON.parse(fs.readFileSync(p, "utf-8"));
}
function getServiceByIdLocal(id) {
  if (id === undefined || id === null) return undefined;
  return loadServicesJSON().find(s => s.id === Number(id));
}

Now, inside your handler function for /api/availability, replace your current hours logic with this block (leave your getOpenSlots(...) call and response structure intact):

const to   = new Date(String(req.query.to));

  return res.status(400).json({ error: "INVALID_QUERY" });
}

let hours;
// Priority: explicit hours -> infer from serviceId -> default 2

  hours = Number(req.query.hours);

  const svc = getServiceByIdLocal(Number(req.query.serviceId));
  hours = (svc && [2,4,6].includes(Number(svc.suggestedHours))) ? Number(svc.suggestedHours) : 2;
} else {
  hours = 2;
}

// Optional: small travel/setup buffer so back-to-back bookings don't collide
